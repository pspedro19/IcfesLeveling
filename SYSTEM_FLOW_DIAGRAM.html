def calculate_weaknesses(answers): weaknesses = [] # Group by topic for topic in topics: accuracy = correct_count / total_count avg_time = sum(times) / len(times) if accuracy < 0.4: # Less than 40% priority = 'HIGH' elif accuracy < 0.6: priority = 'MEDIUM' else: priority = 'LOW' if avg_time > expected_time * 1.5: priority = upgrade(priority) weaknesses.append({ 'topic': topic, 'accuracy': accuracy, 'priority': priority, 'videos_needed': True }) return weaknesses

Matching Algorithm

def get_videos_for_weakness(topic, user_level): # 1. Topic similarity (fuzzy matching) topic_match = fuzz.ratio(video.topic, weakness_topic) # 2. Combined relevance score relevance = ( topic_match * 0.4 + # 40% topic match level_match * 0.2 + # 20% level match quality * 0.3 + # 30% quality engagement * 0.1 # 10% engagement ) return videos.order_by(relevance).limit(3)
